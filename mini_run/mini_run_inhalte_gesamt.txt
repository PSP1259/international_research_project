def now_ts():
    return time.strftime("%Y-%m-%d %H:%M:%S")                       # Timestamp: "2026-01-05 13:39:56"


def apply_api_result(row: dict, res: dict) -> None:
    """
    Übernimmt Messwerte aus der API Antwort in die Log Row.
    Damit run_tool lesbarer bleibt.
    """
    row["latency_s"] = res.get("latency_s")
    row["token_out"] = res.get("token_out")
    row["attempt"] = res.get("attempt")
    row["error"] = res.get("error")
    row["error_type"] = res.get("error_type")
    row["retryable"] = res.get("retryable")
    row["retry_after_s"] = res.get("retry_after_s")


def run_tool(client, prompt: str, row: dict, labels: list[str], infer_store, fail_store):
    """
    Einheitlicher Ablauf pro Tool Call.

    1) client.infer(prompt)
    2) Messwerte in row übernehmen
    3) Wenn ok False, Failure loggen
    4) Wenn ok True, JSON extrahieren
    5) Wenn parse failed, Failure loggen
    6) Wenn parse ok, answer speichern und row schreiben
    """

    # 1) Sendet den Prompt an die API
    res = client.infer(prompt)

    # 2) Übernimmt Messwerte aus der API Antwort
    apply_api_result(row, res)

    # 3) API Call erfolgreich?
    if not res.get("ok"):
        infer_store.append(row)
        fail_store.append(failure_row(row, stage="api_error", raw=res.get("raw")))
        return

    # 4) Rohantwort hole, falls None -> leer lassen
    raw = res.get("raw", "") or ""

    # 5) JSON aus Antwort extrahieren {"answer":"B","confidence":92}
    parsed = extract_json(raw)

    # 6.1) JSON Parsing failed
    if parsed is None:
        row["error"] = "parse_failed"
        row["error_type"] = "parse"
        row["retryable"] = False
        infer_store.append(row)
        fail_store.append(failure_row(row, stage="parse_failed", raw=raw))
        return

    # 6.2) JSON Parsing erfolgreich
    ans = parsed.get("answer")
    row["answer"] = ans
    row["conf_self"] = parsed.get("confidence")
    row["answer_valid"] = bool(ans in labels)       # prüft ob Antwort erlaubt ist
    infer_store.append(row)

from pathlib import Path

from lib.io_jsonl import JsonlStore
from lib.mmlu import load_csv, parse_options, gold_letter


BASE_DIR = Path(__file__).resolve().parent

CSV_PATH = BASE_DIR.parent / "mmlu_pro_all_questions.csv"
INFER_JSONL = BASE_DIR / "mini_infer.jsonl"
OUT_JSONL = BASE_DIR / "mini_scored.jsonl"


def main():

    # 1) Lade die Modell Resultate (Predictions) aus mini_infer.jsonl
    infer_rows = JsonlStore(INFER_JSONL).read_all()

    # 2) Sammle alle question_ids, die in den Predictions vorkommen
    needed_ids = set()
    for row in infer_rows:                              # Loop für jede Prediction aus dem Log
        if "question_id" in row:
            needed_ids.add(int(row["question_id"]))     # ID Str -> Int

    # 3) Lade das CSV und baue eine Map: question_id -> gold_answer_letter
    df = load_csv(CSV_PATH)
    gold_by_qid = {}

    for _, r in df.iterrows():             # Loop für jede Zeile des df CSV
        qid = int(r["question_id"])        # IDs sammeln

        # Wenn diese Frage nicht im Set der Predictions ist -> überspringen
        if qid not in needed_ids:
            continue

        options = parse_options(r["options_json"])              # JSON str -> Python Liste
        answer_index = int(r["answer_index"])                   # Option als Zahl korrekt?
        gold_by_qid[qid] = gold_letter(answer_index, options)   # Speichert Buchstaben in die Map       

    # 4) Löscht altes scored File, erstellt neues
    if OUT_JSONL.exists():
        OUT_JSONL.unlink()
    out_store = JsonlStore(OUT_JSONL)

    # 5) Jede Prediction bewerten und speichern
    for row in infer_rows:
        qid = int(row["question_id"])
        pred = row.get("answer")
        gold = gold_by_qid.get(qid)
        row["label_answer_letter"] = gold

        if gold is None or pred is None:
            row["correct"] = None
        else:
            row["correct"] = int(pred == gold)  # Vergleicht Prediction mit Gold: 1 oder 0

        out_store.append(row)                   # Schreibt die fertige Zeile ins JSONL File

    print("Wrote:", OUT_JSONL)                  # Informativ fürs Terminal 


if __name__ == "__main__":
    main()

import time
import os
import json
from openai import OpenAI
from google import genai

from .errors import ConfigError


MAX_RETRIES = 5          # Max. Wiederholungen
BASE_SLEEP = 1.0         # Start-Wartezeit (Sek.)
MAX_SLEEP = 30.0         # Max. Wartezeit
ERR_LIMIT = 240          # Max. Länge Error-Text


def sleep_backoff(attempt: int):
    seconds = min(MAX_SLEEP, BASE_SLEEP * (2 ** (attempt - 1)))
    time.sleep(seconds)
"""
Exponentialer Backoff pro Versuch.
attempt 1 -> 1.0 Sekunde
attempt 2 -> 2.0 Sekunden
attempt 3 -> 4.0 Sekunden
Wartezeit wächst mit Potenz von 2.
min(MAX_SLEEP, ...) setzt eine Obergrenze von 30.0 Sekunden.
"""


def extract_json(text: str):
    if not text:
        return None

    text = text.strip()
    try:
        return json.loads(text)
    except Exception:
        pass

    # Try to extract the first JSON object from surrounding text.
    start = text.find("{")
    if start == -1:
        return None

    depth = 0
    for i in range(start, len(text)):
        ch = text[i]
        if ch == "{":
            depth += 1
        elif ch == "}":
            depth -= 1
            if depth == 0:
                candidate = text[start : i + 1]
                try:
                    return json.loads(candidate)
                except Exception:
                    return None

    return None


def _openai_temperature(model: str):
    # Some models (e.g. gpt-5-nano) only accept the default temperature.
    if model.startswith("gpt-5"):
        return None
    return 0.0


def _gemini_text(resp) -> str:
    text = getattr(resp, "text", None)
    if text:
        return text

    candidates = getattr(resp, "candidates", None)
    if candidates:
        cand0 = candidates[0]
        content = getattr(cand0, "content", None)
        if content is not None:
            content_text = getattr(content, "text", None)
            if content_text:
                return content_text
            parts = getattr(content, "parts", None)
            if parts:
                texts = []
                for part in parts:
                    part_text = getattr(part, "text", None)
                    if part_text:
                        texts.append(part_text)
                if texts:
                    return "".join(texts)

    choices = getattr(resp, "choices", None)
    if choices:
        try:
            return choices[0].message.content
        except Exception:
            pass

    return ""


# OpenAI Client:
class OpenAIClientEasy:
    def __init__(self, model: str):
        self.client = OpenAI()      # Holt Key selbständig
        self.model = model

    # OpenAI SDK => Role Based Prompting (System - User) -> chat.completions.create()
    def infer(self, prompt: str) -> dict:
        for attempt in range(1, MAX_RETRIES + 1):
            t0 = time.time()
            try:
                req = {
                    "model": self.model,                          # welches modell soll antworten
                    "messages": [
                        # System Message
                        {"role": "system", "content": "Return only the final answer."},
                        # User Message
                        {"role": "user", "content": prompt},
                    ],
                }
                temp = _openai_temperature(self.model)
                if temp is not None:
                    req["temperature"] = temp

                resp = self.client.chat.completions.create(**req)    # API request

                return {
                    "ok": True,
                    "attempt": attempt,
                    "latency_s": round(time.time() - t0, 3),
                    "raw": resp.choices[0].message.content,
                    "token_out": getattr(getattr(resp, "usage", None), "completion_tokens", None),
                    "error": None,
                    "error_type": None,
                    "retryable": None,
                    "retry_after_s": None,
                }

            except Exception as e:
                if attempt == MAX_RETRIES:
                    return {
                        "ok": False,
                        "attempt": attempt,
                        "latency_s": round(time.time() - t0, 3),
                        "raw": None,
                        "token_out": None,
                        "error": str(e)[:ERR_LIMIT],
                        "error_type": type(e).__name__,
                        "retryable": None,
                        "retry_after_s": None,
                    }

                sleep_backoff(attempt)

class GeminiClientEasy:
    def __init__(self, model: str):
        key = os.getenv("GOOGLE_API_KEY")             # Ich muss den Key selbst holen
        if not key:
            raise ConfigError("GOOGLE_API_KEY missing")

        self.client = genai.Client(api_key=key)
        self.model = model

    # Gemini SDK => Single Prompt API -> models.generate_content()
    def infer(self, prompt: str) -> dict:
        for attempt in range(1, MAX_RETRIES + 1):
            t0 = time.time()
            try:
                resp = self.client.models.generate_content(
                    model=self.model,
                    contents=prompt,
                    config=genai.types.GenerateContentConfig(
                        temperature=0.0,
                    )
                )

                return {
                    "ok": True,
                    "attempt": attempt,
                    "latency_s": round(time.time() - t0, 3),
                    "raw": _gemini_text(resp),
                    "token_out": None,
                    "error": None,
                    "error_type": None,
                    "retryable": None,
                    "retry_after_s": None,
                }

            except Exception as e:
                if attempt == MAX_RETRIES:
                    return {
                        "ok": False,
                        "attempt": attempt,
                        "latency_s": round(time.time() - t0, 3),
                        "raw": None,
                        "token_out": None,
                        "error": str(e)[:ERR_LIMIT],
                        "error_type": type(e).__name__,
                        "retryable": None,
                        "retry_after_s": None,
                    }

                sleep_backoff(attempt)

import os
import json
from pathlib import Path
import hashlib

from .errors import ConfigError, APIError, DataError

CACHE_PATH = Path(__file__).resolve().parent / "deepl_cache.json"
USAGE_PATH = Path(__file__).resolve().parent / "deepl_usage.json"

PERSONAL_CHAR_LIMIT = 20_000  # DeepL monthly Limit is 500'000

def _load_json(path: Path, default):
    if not path.exists():
        return default
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except Exception as e:
        raise DataError(f"Could not read {path} ({e})")

def _save_json(path: Path, obj):
    try:
        path.write_text(json.dumps(obj, ensure_ascii=False, indent=2), encoding="utf-8")
    except Exception as e:
        raise DataError(f"Could not write {path} ({e})")

def _key(text: str, target_lang: str) -> str:
    h = hashlib.sha1((target_lang + "|" + text).encode("utf-8")).hexdigest()
    return h

def _get_used_chars() -> int:
    usage = _load_json(USAGE_PATH, {"used_chars": 0})
    return int(usage.get("used_chars", 0))

def _add_used_chars(n: int) -> None:
    usage = _load_json(USAGE_PATH, {"used_chars": 0})
    usage["used_chars"] = int(usage.get("used_chars", 0)) + int(n)
    _save_json(USAGE_PATH, usage)

def translate_text(text: str, target_lang: str, source_lang: str = "EN") -> str:
    """
    Translates text using DeepL with:
    - caching
    - monthly character budget protection
    """
    text = text or ""
    if not text.strip():
        return text

    auth = os.getenv("DEEPL_API_KEY")
    if not auth:
        raise ConfigError("DEEPL_API_KEY missing")

    cache = _load_json(CACHE_PATH, {})
    k = _key(text, target_lang)
    if k in cache:
        return cache[k]

    # Budget check BEFORE calling API
    used = _get_used_chars()
    needed = len(text)
    if used + needed > PERSONAL_CHAR_LIMIT:
        # No translation, return original (or raise if you prefer)
        return text

    try:
        import deepl
        translator = deepl.Translator(auth)
        res = translator.translate_text(
            text,
            source_lang=source_lang,
            target_lang=target_lang
        )
        out = str(res)
    except Exception as e:
        raise APIError(f"DeepL translate failed ({e})")

    cache[k] = out
    _save_json(CACHE_PATH, cache)
    _add_used_chars(needed)
    return out

def translate_question_and_options(question: str, options: list[str], target_lang: str) -> tuple[str, list[str]]:
    q_t = translate_text(question, target_lang=target_lang)
    opts_t = [translate_text(o, target_lang=target_lang) for o in options]
    return q_t, opts_t

import json

class JsonlStore:      
    def __init__(self, path: Path):     # Erstellt Unter-/Ordner auch wenn bereits existiert
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)

    def append(self, row):              # Sicherheitscheck -> verhindert kaputte JSON-Zeilen
        if row is None:
            return

        try:
            with self.path.open("a", encoding="utf-8") as f:         # "a" append (ans Ende hängen)
                f.write(json.dumps(row, ensure_ascii=False) + "\n")  # jede Zeile ein Objekt
        except Exception as e:
            raise DataError(f"Could not write JSONL: {self.path} ({e})")

    def read_all(self):                     # Output immer eine Liste von Dictionaries
        if not self.path.exists():          # Wenn Datei nicht existiert => leere Liste
            return []

        try:
            with self.path.open("r", encoding="utf-8") as f:     # "r" Lesemodus
                result = []                                      # sammelt alle Datensätze
                for line in f:                                   # Datei wird zeilenweise gelesen
                    if line.strip():                             # ignoriert leere Zeilen
                        result.append(json.loads(line))     
                return result
        except Exception as e:
            raise DataError(f"Could not read JSONL: {self.path} ({e})")

"""
load_csv: liest dein MMLU-CSV als DataFrame
clean_text: säubert Frage-Text (whitespace, control chars)
parse_options: macht aus options_json (String) -> Liste von Optionen

CSV
 └─ load_csv()
      └─ question row
           └─ parse_options()
                └─ labels_for()
                     └─ gold_letter()
"""

import json
import re
from pathlib import Path

import pandas as pd

from .errors import DataError


# Hat das CSV diese Spalten?
NEEDED_COLS = ["question_id", "category", "question", "options_json", "answer_index", "src"]


def clean_text(s) -> str:       # Basic cleanup for text

    if s is None:
        return ""

    # statt crash bei None => leere Zelle ""
    s = str(s)                 

    # remove control characters
    s = re.sub(r"[\x00-\x08\x0B\x0C\x0E-\x1F\x7F]", "", s) 

    # Mehrere Spaces/Tabs werden zu einem Space
    s = re.sub(r"[ \t]+", " ", s)

    # Wenn es 3+ neue Zeilen hast → macht nur 2
    s = re.sub(r"\n{3,}", "\n\n", s)

    # Trimmt vorne/hinten whitespace
    return s.strip()


def parse_options(options_json: str) -> list[str]:  # Options: ["Paris", "London", "Berlin"]

    try:
        opts = json.loads(options_json)             # String → Python-Objekt
    except Exception:
        return []

    if not isinstance(opts, list):                  # Muss eine Liste sein
        return []

    return [clean_text(x) for x in opts]            # text cleanen


# Dynamisches Setup statt hardcoden
def labels_for(options: list[str]) -> list[str]:    # Zu jeder Option ein Buchstaben erzeugen
    labels = []                                     # erzeugt eine Liste
    for i in range(len(options)):                   # 0, 1, 2...
        labels.append(chr(ord("A") + i))            # "A", "B", "C"...
    return labels


# Antwort prüfen
def gold_letter(answer_index: int, options: list[str]): 
    if 0 <= answer_index < len(options):                    # Antwortindex muss höher als 0 sein
        return chr(ord("A") + answer_index)                 # Buchstaben für Antwort zurückgeben
    return None


def load_csv(csv_path: str | Path) -> pd.DataFrame:
    csv_path = Path(csv_path)

    # Datei gefunden?
    try:
        df = pd.read_csv(csv_path, encoding="utf-8")                
    except Exception as e:
        raise DataError(f"Could not read CSV: {csv_path} ({e})") 

    # Spalten prüfen
    missing = []
    for c in NEEDED_COLS:
        if c not in df.columns:
            missing.append(c)

    # Abbruch bei Fehler
    if missing:
        raise DataError(f"CSV missing columns: {missing}")

    return df

import hashlib
from pathlib import Path

from .mmlu import labels_for


def load_text(path: str | Path) -> str:
    return Path(path).read_text(encoding="utf-8")


def sha1_10(text: str) -> str:
    return hashlib.sha1(text.encode("utf-8")).hexdigest()[:10]


def build_prompt(template: str, question: str, options: list[str]) -> tuple[str, list[str]]:
    labels = labels_for(options)
    options_block = "\n".join(f"{label}. {option}" for label, option in zip(labels, options))
    prompt = template.format(
        question=question,
        options=options_block,
        labels=",".join(labels),
    )
    return prompt, labels
