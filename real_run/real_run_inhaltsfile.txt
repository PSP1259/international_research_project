  "math": 94,

def create_sample():
  print("---Start Sampling Process---")

  # 1) Check ob INPUT_FILE da ist
  if not INPUT_FILE.exists():
    print(f"FEHLER: Konnte {INPUT_FILE} nicht finden!")
    return  # << wichtig

  # 2) Check ob CSV lesbar ist
  print(f"Lade Quelldatei: {INPUT_FILE} ...")
  try:
    df_all = pd.read_csv(INPUT_FILE, encoding="utf-8")
  except Exception as e:
    print(f"Fehler beim Lesen der CSV: {e}")
    return  # << wichtig

  print(f"Gesamtanzahl Fragen im Original: {len(df_all)}")

  sampled_frames = []

  # 3) Loop durch die Kategorien
  print("Ziehe zufällige Fragen pro Kategorie...")
  for category_name, count in SAMPLE_CONFIG.items():
    df_category = df_all[df_all["category"] == category_name]
    available = len(df_category)

    if available < count:
      print(f"WARNUNG: {category_name}: Wollte {count}, habe nur {available}. Nehme alle.")
      count = available

    if available > 0 and count > 0:
      sample = df_category.sample(n=count, random_state=49)
      sampled_frames.append(sample)
      print(f"Check: {category_name} : {len(sample)}")
    else:
      print(f"Check: {category_name} : 0 (keine Daten)")

  # 4) Speichern
  if sampled_frames:
    final_df = pd.concat(sampled_frames, ignore_index=True)
    OUTPUT_FILE.parent.mkdir(parents=True, exist_ok=True)

    final_df.to_csv(OUTPUT_FILE, index=False, encoding="utf-8")  # << DAS hat gefehlt
    print(f"Sample gespeichert in: {OUTPUT_FILE}")
    print(f"Anzahl Fragen im Sample: {len(final_df)}")
  else:
    print("Fehler: Kein Sample erzeugt (sampled_frames ist leer).")

if __name__ == "__main__":
  create_sample()

import os
import json
import time
import pandas as pd
import deepl
from pathlib import Path
from dotenv import load_dotenv

load_dotenv()

BASE_DIR = Path(__file__).resolve().parent
INPUT_FILE = BASE_DIR / "data" / "mmlu_sample_englisch.csv"
OUTPUT_FILE = BASE_DIR / "data" / "mmlu_ready_bilingual.csv" 

CHAR_LIMIT_WARNING = 450000     # Monatliche Limit: 500'000


def translate_run():
  print("---Starte Übersetzungsprozess---")

  # 1. API Key holen
  api_key = os.getenv("DEEPL_API_KEY")
  if not api_key:
    print("FEHLER: Kein DEEPL_API_KEY in der .env gefunden!")
    return

  # 2. Translater initialisieren
  translator = deepl.Translator(api_key)

  # Check Usage
  try:
    usage = translator.get_usage()
    if usage.character.limit_reached:
      print("WARNUNG: DeepL Limit erreicht!")
      return
    print(f"DeepL Status: {usage.character.count} / {usage.character.limit} Zeichen genutzt.")
  except Exception as e:
    print(f"Warnung: Konnte DeepL Status nicht abrufen: {e}")

  # 3. Datei laden (mmlu_sample_english.csv)
  df = pd.read_csv(INPUT_FILE)
  print(f"Lade {len(df)} Fragen zur Übersetzung...")

  # Neue Spalten vorbereiten
  df["question_de"] = ""
  df["options_de_json"] = ""

  # Zähler für verbrauchte Zeichen
  chars_used_session = 0

  # 4. Translator Loop
  print("---Starte Translator Loop---")

  start_time = time.time()

  # a) Frage übersetzen EN -> DE
  for index, row in df.iterrows():

    # 1. Variablen initialisieren
    q_de = ""
    options_json_de = "[]"

    # A) Frage übersetzen
    q_en = row["question"]

    # Sicherheits-Check: Ist die englische Frage leer?
    if pd.isna(q_en) or str(q_en).strip() == "":
      print(f"Warnung: Leere Frage bei ID {row.get('question_id')}. Überspringe.")
      continue

    # Notbremse
    if chars_used_session > CHAR_LIMIT_WARNING:
      print("NOTBREMSE: Zeichen-Limit nähert sich!")
      break

    try:
      # DeepL Call für die Frage
      res_q = translator.translate_text(str(q_en), target_lang="DE")
      q_de = res_q.text
      chars_used_session += len(str(q_en))
    except Exception as e:
      print(f"Fehler bei Frage ID {row.get('question_id')}: {e}")
      # Wenn die Frage nicht übersetzt werden kann => überspringen!
      continue 

    # B) Optionen übersetzen
    try:
      # 1. String "[...]" in echte Python-Liste umwandeln
      options_list_en = json.loads(row["options_json"])

      # 2. Liste an DeepL senden
      res_opts = translator.translate_text(options_list_en, target_lang="DE")

      # 3. Ergebnisse extrahieren
      if isinstance(res_opts, list):
        options_list_de = [r.text for r in res_opts]
      else:
        options_list_de = [res_opts.text]

      # 4. Zurück in JSON-String wandeln
      options_json_de = json.dumps(options_list_de, ensure_ascii=False)

      # Zeichen zählen
      for opt in options_list_en:
        chars_used_session += len(str(opt))

    except Exception as e:
      print(f"Fehler bei Optionen ID {row.get('question_id')}: {e}")
      # Fallback: Speichert eine leere Liste, damit das Skript nicht abstürzt
      options_json_de = "[]"

    # C) Ins DataFrame schreiben (Jetzt ist q_de sicher definiert!)
    df.at[index, "question_de"] = q_de
    df.at[index, "options_de_json"] = options_json_de

    # Fortschrittsanzeige alle 50 Zeilen
    if (index + 1) % 50 == 0:
      print(f"   ... {index + 1} / {len(df)} erledigt. (Ca. {chars_used_session} Zeichen verbraucht)")

  # 6. Speichern
  output_dir = OUTPUT_FILE.parent
  output_dir.mkdir(parents=True, exist_ok=True)

  df.to_csv(OUTPUT_FILE, index=False)

  duration = time.time() - start_time
  print("-" * 30)
  print(f"FERTIG! Gespeichert unter: {OUTPUT_FILE}")
  print(f"Dauer: {round(duration, 2)} Sekunden")
  print(f"Zeichen in dieser Session: {chars_used_session}")
  print("-" * 30)


if __name__ == "__main__":
  translate_run()

import json
import pandas as pd
from pathlib import Path

# Pfade
BASE_DIR = Path(__file__).resolve().parent
RESULTS_DIR = BASE_DIR / "results"
INPUT_FILE = BASE_DIR / "data" / "mmlu_ready_bilingual.csv"



def check():
  print("\n--- MMLU RUN STATUS MONITOR --- ")
    


  # 1. Ziel setzen
  if INPUT_FILE.exists():
    df = pd.read_csv(INPUT_FILE)
    total_goal = len(df)
    print(f"🏁 ZIEL: {total_goal} Fragen pro Modell/Sprache\n")
  else:
    print("Warnung: Input-Datei nicht gefunden. Ziel unbekannt.\n")
    total_goal = 0

  if not RESULTS_DIR.exists():
    print("Noch keine Ergebnisse vorhanden.")
    return




  # 2. Ergebnisse scannen
  files = sorted(list(RESULTS_DIR.glob("*.jsonl")))

  if len(files) == 0:
    print("Noch keine .jsonl Dateien im results Ordner gefunden.")
    return
    
  # Tabellenkopf
  print(f"{'DATEI NAME':<40} | {'FORTSCHRITT':<15} | {'STATUS'}")
  print("-" * 75)

  for f in files:
    count = 0

    # Robust zählen: nur gültige JSON Zeilen zählen, die question_id enthalten
    try:
      with open(f, "r", encoding="utf-8") as file:
        for line in file:
          line = line.strip()
          if not line:
            continue
          try:
            obj = json.loads(line)
            if "question_id" in obj:
              count += 1
          except:
            # Kaputte Zeilen ignorieren, damit Fortschritt nicht verfälscht wird
            continue
    except:
      count = 0
        
    # Prozent berechnen
    percent = 0
    if total_goal > 0:
      percent = (count / total_goal) * 100
        
    # Status
    if count >= total_goal and total_goal > 0:
      status = "FERTIG"
    elif count == 0:
      status = "STARTBEREIT"
    else:
      status = f"🏃 {percent:.1f}%"

    print(f"{f.name:<40} | {count}/{total_goal:<5}       | {status}")

  print("-" * 75)
  print("Starte Jobs im Ordner 'jobs/', um weiterzumachen.\n")

if __name__ == "__main__":
    check()

===== FILE: lib\ai_client.py =====
import os
import json
import logging
import math

from google import genai
from google.genai import types
from openai import OpenAI
from dotenv import load_dotenv


# Einheitliches Logging-Verhalten
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("AIClient")


# Universal-Client
class AIClient:

  """
  - Konstruktor: Wird aufgerufen sobald ein neues Objekt erstellt wird.

  - *Args: Not sure how many arguments might be passed to a function
      - provider (str): "google" oder "openai"
      - model_name (str): z.B. "chatgpt-5.2" oder "gemini-2.0-flash"
      - is_reasoning (bool): Wenn True -> aktivieren
  """

  def __init__(self, provider: str, model_name: str, is_reasoning: bool = False):

    # Ladet die API Keys neu
    load_dotenv()

    # Speichert die Einstellungen in "self"
    self.provider = provider.lower()
    self.model_name = model_name
    self.is_reasoning = is_reasoning
    self._google_logprobs_supported = True

    # Setup der Clients
    if self.provider == "google":
      api_key = os.getenv("GOOGLE_API_KEY")
      if not api_key:
        raise ValueError("CRITICAL: GOOGLE_API_KEY fehlt in .env!")
      # HttpOptions.timeout is in milliseconds.
      http_options = types.HttpOptions(timeout=60000)
      self.client = genai.Client(api_key=api_key, http_options=http_options)
      # Gemini 3 preview models do not support logprobs in this setup.
      if self.model_name in {"gemini-3-flash-preview", "gemini-3-pro-preview"}:
        self._google_logprobs_supported = False

    elif self.provider == "openai":
      api_key = os.getenv("OPENAI_API_KEY")
      if not api_key:
        raise ValueError("CRITICAL: OPENAI_API_KEY fehlt in .env!")
      self.client = OpenAI(api_key=api_key)

    else:
      raise ValueError(f"Unbekannter Provider: {provider}")

    logger.info("Client gestartet: %s - %s", self.provider.upper(), self.model_name)


  def infer(self, prompt: str):
    try:
      if self.provider == "google":
        return self._call_google(prompt)
      elif self.provider == "openai":
        return self._call_openai(prompt)
    except Exception as e:
      logger.error(f"Fehler im API Call: {e}")
      return None, None
    

  def _calculate_prob(self, logprob_val):
  # Hilfsfunktion: Wandelt logprob (-0.1) in Prozent (90.4) um
    try:
      if logprob_val is None: return None
      # e hoch logprob = Wahrscheinlichkeit (0.0 bis 1.0)
      return math.exp(logprob_val) * 100
    except:
      return None


  def _extract_top_choices(self, target_char, logprobs_data, provider):
    """
    Sucht den Token, der die Antwort (A, B, C...) enthält, und holt dessen Konkurrenten.
    """
    if not target_char or not logprobs_data:
      return []

    found_choices = []

    def _clean_token_text(text):
      if text is None:
        return None
      t = str(text).strip()
      if len(t) >= 2 and t[0] == t[-1] and t[0] in ("\"", "'"):
        t = t[1:-1]
      return t

    def _candidate_text(candidate):
      if candidate is None:
        return None
      for attr in ("token", "text"):
        val = getattr(candidate, attr, None)
        if isinstance(val, str) and val.strip():
          return val
      content = getattr(candidate, "content", None)
      parts = getattr(content, "parts", None)
      if parts and len(parts) > 0:
        text = getattr(parts[0], "text", None)
        if isinstance(text, str) and text.strip():
          return text
      token_id = getattr(candidate, "token_id", None)
      if isinstance(token_id, str) and token_id.strip():
        return token_id
      return None
    
    
    
    # --- OPENAI LOGIC ---

    try:
      if provider == "openai":
        # logprobs_data ist hier: response.choices[0].logprobs.content
        for token_data in logprobs_data:
          # Bereinigen (Leerzeichen entfernen) und vergleichen
          clean_token = token_data.token.strip()
          
          # Wenn wir den Antwort-Buchstaben finden (z.B. "A")
          if clean_token == target_char:
            # Top-Logprobs (die Alternativen) für DIESEN Token
            if token_data.top_logprobs:
              for alt in token_data.top_logprobs:
                prob_percent = math.exp(alt.logprob) * 100
                found_choices.append({
                  "token": alt.token,
                  "prob": round(prob_percent, 2)
                })
            break # Haben ihn gefunden, Loop beenden

    # --- GOOGLE LOGIC ---
      elif provider == "google":
        # Google liefert top_candidates pro Schritt. 
        # wo der gewählte Kandidat unser Buchstabe ist.
        # logprobs_data ist hier: response.candidates[0].logprobs_result
        
        chosen_tokens = logprobs_data.chosen_candidates
        top_candidates_list = logprobs_data.top_candidates

        for i, chosen in enumerate(chosen_tokens):
          chosen_text = _candidate_text(chosen)
          if not chosen_text:
            continue
          if _clean_token_text(chosen_text) != target_char:
            continue

          if len(top_candidates_list) > i:
            top_entry = top_candidates_list[i]
            candidates = getattr(top_entry, "candidates", None) or top_entry
            if candidates:
              for cand in candidates:
                cand_text = _candidate_text(cand)
                logp = getattr(cand, "log_probability", None)
                if cand_text is None or logp is None:
                  continue
                p = math.exp(logp) * 100
                found_choices.append({
                  "token": cand_text,
                  "prob": round(p, 2)
                })
              break
    except Exception as e:
      logger.warning(f"Konnte Top-Choices nicht extrahieren: {e}")
      
    return found_choices           








  # ------------------ GOOGLE ----------------------

  # Gemini SDK => Single Prompt API -> models.generate_content()
  def _call_google(self, prompt):
    use_logprobs = self._google_logprobs_supported
    if use_logprobs:
      config = types.GenerateContentConfig(
        temperature=0.0,                              # Deterministisch
        response_mime_type="application/json",
        response_logprobs=True,
        logprobs=4
      )
    else:
      config = types.GenerateContentConfig(
        temperature=0.0,
        response_mime_type="application/json"
      )
    try:
      response = self.client.models.generate_content(
        model=self.model_name,
        contents=prompt,
        config=config
      )




      # 1. Antwort Parsen
      json_resp = json.loads(response.text)
      answer_char = json_resp.get("answer_char")



      # 2. Logprobs extrahieren (Google Logic)
      # Wir nehmen den Durchschnitt der Logprobs aller Tokens im Output
      avg_logprob = None
      top_choices_extracted = []
      if use_logprobs and response.candidates and response.candidates[0].logprobs_result:
        # Hole alle logprobs der gewählten Tokens
        tokens = response.candidates[0].logprobs_result.chosen_candidates
        if tokens:
          # Durchschnitt berechnen
          vals = [t.log_probability for t in tokens if t.log_probability is not None]
          if vals:
            avg_logprob = sum(vals) / len(vals)

        top_choices_extracted = self._extract_top_choices(
          answer_char,
          response.candidates[0].logprobs_result,
          "google"
        )

      # 3. Usage
      usage = {
        "input_tokens": response.usage_metadata.prompt_token_count if response.usage_metadata else 0,
        "output_tokens": response.usage_metadata.candidates_token_count if response.usage_metadata else 0,
        "raw_logprob": avg_logprob,
        "prob_percent": self._calculate_prob(avg_logprob),
        "top_choices": top_choices_extracted
      }

      return json_resp, usage

    except Exception as e:
      if use_logprobs and ("Logprobs is not enabled" in str(e) or "logprobs" in str(e)):
        logger.warning("Google Logprobs nicht unterstuetzt. Deaktiviere fuer diesen Lauf.")
        self._google_logprobs_supported = False
        return self._call_google(prompt)

      logger.error(f"Google Fehler: {e}")
      return None, None





  # ------------------ OpenAI ----------------------
  
  # OpenAI SDK => Role Based Prompting (System - User) -> chat.completions.create()

  def _call_openai(self, prompt):

    # 1. Basis-Parameter
    params = {
      "model": self.model_name,
      "messages": [{"role": "user", "content": prompt}],
      "response_format": {"type": "json_object"},
      "logprobs": True,
      "top_logprobs": 4
    }

    # 2. Reasoning True/False?
    if self.is_reasoning:
      # ---FALL A: REASONING AKTIV---
      # Temperature ist hier VERBOTEN (fuehrt zu API Error).
      # none (default), low, medium, high, xhigh
      params["reasoning_effort"] = "xhigh"

    else:
      # ---FALL B: STANDARD / KEIN REASONING---
      # Das gilt fuer GPT-4.1 UND fuer GPT-5.2 im Modus "none".
      # Hier MUESSEN wir Temperature 0 setzen fuer reproduzierbare Ergebnisse.
      params["temperature"] = 0.0

      # Speziell fuer GPT-5.2: Wenn wir 'is_reasoning=False' haben,
      # sagen wir der API explizit: "Bitte nicht nachdenken".
      if self.model_name.startswith("gpt-5"):
        params["reasoning_effort"] = "none"

    # 3. Der Call
    try:
      # **params entpackt das Dictionary automatisch in die richtigen Argumente
      response = self.client.chat.completions.create(**params)

      # 1. Antwort extrahieren & JSON Parsen
      content = response.choices[0].message.content
      json_response = json.loads(content) # Wir brauchen das JSON jetzt schon!
      answer_char = json_response.get("answer_char") # Wonach wir suchen (z.B. "A")

      # 2. Logprobs Analyse
      avg_logprob = None
      top_choices_extracted = [] # <--- NEU
      
      if response.choices[0].logprobs and response.choices[0].logprobs.content:
        tokens = response.choices[0].logprobs.content
        
        # Durchschnitt (wie bisher)
        vals = [t.logprob for t in tokens if t.logprob is not None]
        if vals: avg_logprob = sum(vals) / len(vals)

        # <--- NEU: Die Alternativen holen
        top_choices_extracted = self._extract_top_choices(answer_char, tokens, "openai")


      # 3. Usage
      u = response.usage
      usage = {
        "input_tokens": u.prompt_tokens if u else 0,
        "output_tokens": u.completion_tokens if u else 0,
        "raw_logprob": avg_logprob,
        "prob_percent": self._calculate_prob(avg_logprob),
        "top_choices": top_choices_extracted  # <--- HIER SPEICHERN
      }

      return json_response, usage



    except Exception as e:
    # Fallback: Wenn Logprobs nicht erlaubt sind 
      if "logprobs" in str(e) and params.get("logprobs"):
        logger.warning("Logprobs nicht unterstuetzt. Versuche ohne...")
        params.pop("logprobs")
        params.pop("top_logprobs")
        try:
          response = self.client.chat.completions.create(**params)
          content = response.choices[0].message.content
          u = response.usage
          usage = {
            "input_tokens": u.prompt_tokens if u else 0,
            "output_tokens": u.completion_tokens if u else 0,
            "raw_logprob": None,
            "prob_percent": None
          }
          return json.loads(content), usage
        except Exception as e2:
          logger.error(f"OpenAI Retry failed: {e2}")
          return None, None
            
      logger.error(f"OpenAI Fehler: {e}")
      return None, None

import pandas as pd
import json
import time
import os
from pathlib import Path
from lib.ai_client import AIClient


current_file_path = Path(__file__).resolve()

BASE_DIR = current_file_path.parent.parent  

INPUT_FILE = BASE_DIR / "data" / "mmlu_ready_bilingual.csv"


# ----- Prompt Templates -----

PROMPT_TEMPLATES = {
  "EN": """
  Question: {question}
  Options:
  {options}

  Select the correct option letter (A, B, C, ...).
  Return ONLY a JSON object using this format (replace values with your result):
  {{
    "answer_char": "X",
    "confidence": 0
  }}
  """,

  "DE": """
  Frage: {question}
  Optionen:
  {options}

  Waehle den korrekten Buchstaben (A, B, C, ...).
  Antworte NUR mit einem JSON-Objekt in diesem Format (ersetze die Platzhalter!):
  {{
    "answer_char": "X",
    "confidence": 0
  }}
  """
}


def _parse_options(opts_text):
  try:
    opts = json.loads(opts_text)
  except Exception:
    return []
  if not isinstance(opts, list):
    return []
  return [str(x).strip() for x in opts]


def _labels_for(options):
  return [chr(ord("A") + i) for i in range(len(options))]


def _format_options(options, labels):
  lines = []
  for label, option in zip(labels, options):
    lines.append(f"{label}) {option}")
  return "\n".join(lines)


def _normalize_text(value):
  return " ".join(str(value).strip().split()).lower()


def _normalize_label(value):
  if value is None:
    return None
  label = str(value).strip().upper()
  return label if label else None


def _normalize_answer(answer_raw, options, labels):
  if answer_raw is None:
    return None
  raw = str(answer_raw).strip()
  if raw == "":
    return None
  if not labels:
    return raw

  raw_upper = raw.upper()
  for label in labels:
    if raw_upper == label:
      return label

  for label in labels:
    if raw_upper.startswith(label) and len(raw_upper) > 1:
      next_char = raw_upper[1]
      if not next_char.isalnum():
        return label

  norm_raw = _normalize_text(raw)
  for idx, opt in enumerate(options):
    if _normalize_text(opt) == norm_raw:
      return labels[idx]

  return raw


def run_benchmark(config):

  # Dateiname generieren
  filename = f"results_{config['model_name']}_{config['language']}"
  if config["is_reasoning"]:
    filename += "_reasoning"
  filename += ".jsonl"

  OUTPUT_FILE = BASE_DIR / "results" / filename

  # Start Meldung
  print(f"\nJOB: {config['model_name']} [{config['language']}]")
  if config.get("max_questions"): print(f"LIMIT: {config['max_questions']}")



  # 1. Daten laden
  if not INPUT_FILE.exists():
    print(f"Input Datei fehlt: {INPUT_FILE}")
    return

  df = pd.read_csv(INPUT_FILE)
  total_questions = len(df)



  # 2. Resume Logik (Was haben wir schon?)
  completed_ids = set()
  OUTPUT_FILE.parent.mkdir(parents=True, exist_ok=True)

  if OUTPUT_FILE.exists():
    with open(OUTPUT_FILE, "r", encoding="utf-8") as f:
      for line in f:
        try:
          data = json.loads(line)
          completed_ids.add(data["question_id"])
        except:
          continue  # Kaputte Zeilen ignorieren
    print(f"Resume: {len(completed_ids)} bereits erledigt.")

  # 3. Client initialisieren
  try:
    bot = AIClient(
      provider=config["provider"],
      model_name=config["model_name"],
      is_reasoning=config["is_reasoning"]
    )
  except Exception as e:
    print(f"Client Crash: {e}")
    return

  # 4. Der Loop
  processed_count = 0
  max_q = config.get("max_questions", 999999)  # Default: kein Limit

  # Datei im Append-Modus öffnen
  with open(OUTPUT_FILE, "a", encoding="utf-8") as f_out:

    for index, row in df.iterrows():

      # LIMIT CHECK (Emergency Break)
      if processed_count >= max_q:
        print(f"Tages-Limit von {max_q} Fragen erreicht. Beende Job.")
        break

      q_id = row["question_id"]

      # Skip wenn schon erledigt
      if q_id in completed_ids:
        continue

      lang = config["language"]

      # Text je nach Sprache wählen
      if lang == "DE":
        q_text = row.get("question_de", "")
        opts_text = row.get("options_de_json", "[]")
      else:
        q_text = row.get("question", "")
        opts_text = row.get("options_json", "[]")

      options = _parse_options(opts_text)
      labels = _labels_for(options) if options else []
      options_block = _format_options(options, labels) if options else str(opts_text)

      # Sicherheits-Check: Leere Fragen skippen
      if pd.isna(q_text) or str(q_text).strip() == "":
        print(f"Warnung: Leerer Text für ID {q_id}. Skipping.")
        continue

      # Prompt bauen
      prompt = PROMPT_TEMPLATES[lang].format(
        question=q_text,
        options=options_block
      )

      print(f"ID {q_id} ({index + 1}/{total_questions}) ... ", end="", flush=True)
      start_ts = time.time()

      # API Call
      response_data, usage_data = bot.infer(prompt)

      duration = time.time() - start_ts

      # Ergebnis nur speichern, wenn gültiges JSON zurückkommt
      if isinstance(response_data, dict):
        answer_raw = response_data.get("answer_char")
        answer_char = _normalize_answer(answer_raw, options, labels)
        answer_label = _normalize_label(answer_char)
        correct_label = _normalize_label(row.get("answer"))
        correct = 1 if answer_label and correct_label and answer_label == correct_label else 0
        result_entry = {
          "question_id": q_id,
          "category": row.get("category"),
          "language": lang,
          "model": config["model_name"],
          "is_reasoning": config["is_reasoning"],
          "question_len_char": len(q_text),
          "input_tokens": usage_data.get("input_tokens", 0),
          "output_tokens": usage_data.get("output_tokens", 0),
          "logprob": usage_data.get("raw_logprob"),
          "top_choices": usage_data.get("top_choices", []),
          "correct_answer": row.get("answer"),       
          "answer_char": answer_char,
          "correct": correct,
          "confidence": response_data.get("confidence"),
          "duration_s": round(duration, 2),
          "timestamp": time.strftime("%Y-%m-%d %H:%M:%S")
        }

        # Sofort speichern (JSONL)
        f_out.write(json.dumps(result_entry, ensure_ascii=False) + "\n")
        f_out.flush()

        processed_count += 1

        print(f"OK ({duration:.1f}s)")

      else:
        print("Fehlgeschlagen")

      # Rate-Limit Schutz
      time.sleep(config["sleep_seconds"])

  print(f"Job beendet. {processed_count}!")


if __name__ == "__main__":
  run_benchmark(RUN_CONFIG)
